// White
event SprayPaint_On_White
{
    RunEventChild    MainMenuOverride           SprayPaint_Loop_White                              0.0001
}

event SprayPaint_Loop_White
{
    FireCommand      0.0                        "engine sm_spray sprays/spraypaint_m0reWhite"
    RunEvent         SprayPaint_Loop_White      0.03 // spray speed, setting this too fast will kick you from most servers
}
